# Estonia and the Finnish national movement.
# With Finland under a foreign yoke, Estonian money, rifles and volunteers
# can cross the gulf. The fervor is applied to the OWNER of the Finnish
# cores as a country-wide unrest modifier - vanilla events never apply
# province modifiers to foreign-owned provinces directly.

country_event = {
	id = 95562
	title = "EVTNAME95562"
	desc = "EVTDESC95562"

	picture = "Recruits"

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		NOT = { exists = FIN }
		not = { has_country_flag = finnish_nationalists_supported }
		not = { has_country_flag = finnish_nationalists_declined }
		money = 50000
		nationalism_n_imperialism = 1
	}

	mean_time_to_happen = {
		months = 24
	}

	option = {
		name = "EVTOPTA95562"
		ai_chance = {
			factor = 0.5
		}
		set_country_flag = finnish_nationalists_supported
		money = -50000
		prestige = 5
		any_country = {
			limit = {
				OR = {
					owns = 343
					owns = 336
					owns = 345
				}
			}
			relation = {
				who = THIS
				value = -50
			}
			add_casus_belli = {
				target = THIS
				type = cut_down_to_size
				months = 24
			}
			add_country_modifier = {
				name = finnish_nationalist_fervor
				duration = 1825
			}
		}
	}

	option = {
		name = "EVTOPTB95562"
		ai_chance = {
			factor = 1
		}
		set_country_flag = finnish_nationalists_declined
		any_country = {
			limit = {
				OR = {
					owns = 343
					owns = 336
					owns = 345
				}
			}
			relation = {
				who = THIS
				value = 10
			}
		}
	}
}
