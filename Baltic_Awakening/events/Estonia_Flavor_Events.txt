#########################################################################
# With the port thriving, merchants and industrialists demand a
# formal stock exchange to finance Baltic trade ventures.
country_event = {
	id = 95594
	title = "EVTNAME95594"
	desc = "EVTDESC95594"
	picture = "stock_exchange"

	fire_only_once = yes

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		has_country_flag = tallinn_trade_hub
		NOT = { has_country_flag = tallinn_stock_exchange_founded }
		invention = central_banking
		num_of_ports = 1
	}

	mean_time_to_happen = {
		months = 36
	}

	option = {
		name = "EVTOPTA95594"
		set_country_flag = tallinn_stock_exchange_founded
		prestige = 8
		money = 25000
		add_country_modifier = {
			name = local_stock_exchange
			duration = -1
		}
		add_country_modifier = {
			name = capitalist_expansion
			duration = 1825
		}
		349 = {
			add_province_modifier = {
				name = financial_center
				duration = -1
			}
		}
		any_pop = {
			limit = {
				location = 349
				type = capitalists
			}
			militancy = -2
			consciousness = -1
		}
	}
	option = {
		name = "EVTOPTB95594"
		prestige = -3
		add_country_modifier = {
			name = merchant_decline
			duration = 1825
		}
		any_pop = {
			limit = {
				location = 349
				type = capitalists
			}
			militancy = 2
		}
	}
}

	desc = "EVTDESC95597"
