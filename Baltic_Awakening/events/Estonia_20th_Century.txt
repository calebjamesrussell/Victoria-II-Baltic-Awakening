# Estonia: the long twentieth century.
# From the collapse of the Russian Empire to the digital republic:
# the War of Independence, the interwar republic, the Molotov-Ribbentrop
# Pact, the occupations, the Singing Revolution, and rebirth.
# IDs 96000-96099 are reserved for this mod.

##########################################################################
# 96000 - The Empire Collapses: the February Revolution reaches Reval
##########################################################################
country_event = {
	id = 96000
	title = "EVTNAME96000"
	desc = "EVTDESC96000"
	picture = "revolution_02_news_image"

	fire_only_once = yes

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		year = 1917
		owns = 349
		war = yes
	}

	mean_time_to_happen = {
		months = 6
		modifier = {
			factor = 0.5
			any_country = {
				limit = { tag = RUS }
				war_exhaustion = 0.9
			}
		}
	}

	option = {
		name = "EVTOPTA96000"
		set_country_flag = february_revolution_reached_estonia
		prestige = 2
		any_pop = {
			limit = {
				culture = estonian
				OR = {
					type = labourers
					type = soldiers
				}
			}
			consciousness = 2
			militancy = 1
		}
		any_pop = {
			limit = { culture = north_german }
			militancy = 1
		}
	}
	option = {
		name = "EVTOPTB96000"
		set_country_flag = february_revolution_reached_estonia
		any_pop = {
			limit = { culture = estonian }
			consciousness = 1
		}
	}
}

##########################################################################
# 96005 - The Great Depression hits the young republic (1929-1933)
##########################################################################
country_event = {
	id = 96005
	title = "EVTNAME96005"
	desc = "EVTDESC96005"
	picture = "Poor"

	fire_only_once = yes

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		has_country_flag = war_of_independence_won
		NOT = { has_country_flag = estonian_depression_survived }
		year = 1929
	}

	mean_time_to_happen = {
		months = 18
		modifier = {
			factor = 0.5
			any_owned_province = {
				trade_goods = grain
			}
		}
	}

	option = {
		name = "EVTOPTA96005"
		set_country_flag = estonian_depression_survived
		add_country_modifier = {
			name = estonian_depression
			duration = 1825
		}
		any_pop = {
			limit = {
				OR = {
					type = labourers
					type = farmers
				}
			}
			militancy = 2
		}
		any_pop = {
			limit = { type = capitalists }
			militancy = 1
		}
	}
	option = {
		name = "EVTOPTB96005"
		set_country_flag = estonian_depression_survived
		money = 10000
		add_country_modifier = {
			name = estonian_depression
			duration = 1095
		}
		any_pop = {
			limit = {
				OR = {
					type = labourers
					type = farmers
				}
			}
			militancy = 1
		}
	}
}
