# 95913 - The Stockholm Proposal: Scandinavia courts the United Baltic
# Provinces. If Scandinavia has UBD in its sphere and UBD has Swedish as
# an accepted culture, the estates of Stockholm propose the old dream:
# union of the North, with the Baltic shores in it. UBD may accept -
# inheriting Scandinavia and continuing as her - or refuse, in which
# case the North takes it badly and war follows.
##########################################################################
country_event = {
	id = 95913
	title = "EVTNAME95913"
	desc = "EVTDESC95913"
	picture = "Finepeople"
	major = yes
	fire_only_once = yes

	trigger = {
		tag = UBD
		exists = SCA
		SCA = {
			in_sphere = THIS
		}
		accepted_culture = swedish
	}

	mean_time_to_happen = {
		months = 12
	}

	# Option A - the union: UBD inherits Scandinavia and the player
	# continues as the united North
	option = {
		name = "EVTOPTA95913"
		ai_chance = {
			factor = 1
			modifier = {
				factor = 0.5
				war = yes
			}
		}
		SCA = {
			inherit = THIS
		}
		change_tag = SCA
		add_accepted_culture = estonian
		add_accepted_culture = latvian
		set_country_flag = vana_hea_rootsi_union
	}

	# Option B - refusal: Scandinavia is overstepping, and says so with armies
	option = {
		name = "EVTOPTB95913"
		ai_chance = {
			factor = 0
		}
		SCA = {
			country_event = { id = 95914 days = 0 }
		}
		relation = {
			who = SCA
			value = -100
		}
		any_pop = {
			limit = {
				OR = {
					culture = estonian
					culture = latvian
				}
			}
			consciousness = 1
		}
		set_country_flag = refused_stockholm_proposal
	}
}
