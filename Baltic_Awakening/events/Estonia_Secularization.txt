# Estonia Flavor Events: secularization and freethought (1880-1939)
# POPs become atheist organically: the vanilla conversion engine spreads
# the state religion daily once same-culture seed POPs of that religion
# exist in a province. These events steer consciousness/militancy, grant
# province modifiers that boost conversion, and (for the state atheism
# path) change the state religion itself, which the engine then spreads.
# IDs 95541-95560 reserved for this file.
#
# Note: pop religion cannot be set directly by effects; only the state
# religion can be changed (country scope). Seed POPs are defined in
# history/pops/1836.1.1/Estonia.txt so conversion has valid targets.

#########################################################################
# 95541 - Darwin in the lecture halls
#########################################################################
country_event = {
	id = 95541
	title = "EVTNAME95541"
	desc = "EVTDESC95541"
	picture = "darwin"

	fire_only_once = yes

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		year = 1879
		empiricism = 1
		owns = 352 # Tartu - the university city
	}

	mean_time_to_happen = {
		months = 36
	}

	option = {
		name = "EVTOPTA95541"
		set_country_flag = darwin_debates_allowed
		352 = {
			add_province_modifier = { name = darwin_debates duration = 3650 }
		}
		any_pop = {
			limit = {
				culture = estonian
				type = clerks
			}
			consciousness = 2
		}
		any_pop = {
			limit = {
				culture = estonian
				type = clergymen
			}
			militancy = 1
		}
		# The unbelievers take heart from the public victory of science
		any_pop = {
			limit = {
				has_pop_religion = atheist
			}
			consciousness = 2
		}
	}
	option = {
		name = "EVTOPTB95541"
		set_country_flag = darwin_debates_banned
		any_pop = {
			limit = {
				culture = estonian
				type = clerks
			}
			consciousness = 1
			militancy = 1
		}
		prestige = -1
	}
}

#########################################################################
# 95544 - The quiet congregations (repeatable spread)
#########################################################################
country_event = {
	id = 95544
	title = "EVTNAME95544"
	desc = "EVTDESC95544"
	picture = "Administration"

	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		has_country_flag = secular_schools
		religion = atheist
	}

	mean_time_to_happen = {
		months = 12

		# The freethought press and the movement's own organizations carry
		# the quiet congregations from town to town much faster
		modifier = {
			factor = 0.5
			has_country_flag = freethought_press_patronized
		}
	}

	option = {
		name = "EVTOPTA95544"
		# The conversion engine works per province: reach every province
		# where unbelievers already live, not just one
		any_owned = {
			limit = {
				owner = { OR = { tag = EST tag = UBD } }
				any_pop = { has_pop_religion = atheist }
				NOT = { has_province_modifier = quiet_congregations }
			}
			add_province_modifier = { name = quiet_congregations duration = 730 }
		}
		any_pop = {
			limit = {
				has_pop_religion = atheist
			}
			consciousness = 1
		}
	}
	option = {
		name = "EVTOPTB95544"
		prestige = 1
		any_pop = {
			limit = {
				has_pop_religion = atheist
			}
			militancy = 1
		}
	}
}

# 95549 - The League of Militant Atheists: the revolution's own church
########################################################################
# A proletarian dictatorship turns its guns on the altars: with the
# state already atheist, the League forces the issue, driving the
# remaining faithful toward the conversion engine's targets.
country_event = {
	id = 95555
	title = "EVTNAME95555"
	desc = "EVTDESC95555"
	picture = "Riot"
	fire_only_once = yes
	trigger = {
		OR = {
			tag = EST
			tag = UBD
		}
		government = proletarian_dictatorship
		religion = atheist
	}
	mean_time_to_happen = {
		months = 6
	}
	option = {
		name = "EVTOPTA95555"
		set_country_flag = militant_atheists_organized
		# The League's lectures and demonstrations reach every province:
		# the engine converts per province, so blanket the country
		any_owned = {
			add_province_modifier = {
				name = quiet_congregations
				duration = -1
			}
		}
		# The faithful are shamed into the party's reading circles;
		# conscious POPs convert faster
		any_pop = {
			limit = {
				NOT = { has_pop_religion = atheist }
			}
			consciousness = 2
			militancy = -1
		}
	}
	option = {
		name = "EVTOPTB95555"
		# Softer line: win them with bread, not with decrees
		set_country_flag = league_soft_line
		any_pop = {
			limit = {
				NOT = { has_pop_religion = atheist }
			}
			militancy = -2
		}
		prestige = 1
	}
}
