# Estonia Flavor Events: secularization and freethought (1880-1939)
# POPs can become atheist through these events; the vanilla conversion
# engine then spreads the faith of the state religion daily, provided
# same-culture seed POPs of the target religion exist in a province.
# IDs 95541-95560 reserved for this file.

#########################################################################
# 95541 - Darwin in the lecture halls
#########################################################################
country_event = {
	id = 95541
	title = "EVTNAME95541"
	desc = "EVTDESC95541"
	picture = "darwin"

	fire_only_once = yes

	trigger = {
		tag = EST
		year = 1879
		empiricism = 1
		owns = 352 # Tartu - the university city
	}

	mean_time_to_happen = {
		months = 36
	}

	option = {
		name = "EVTOPTA95541"
		set_country_flag = darwin_debates_allowed
		any_pop = {
			limit = {
				culture = estonian
				type = clerks
			}
			religion = atheist
			consciousness = 2
		}
		any_pop = {
			limit = {
				culture = estonian
				type = clergymen
			}
			militancy = 1
		}
	}
	option = {
		name = "EVTOPTB95541"
		set_country_flag = darwin_debates_banned
		any_pop = {
			limit = {
				culture = estonian
				type = clerks
			}
			consciousness = 1
			militancy = 1
		}
		prestige = -1
	}
}
